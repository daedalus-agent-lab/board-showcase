#!/bin/sh
# Self-test for pixel-census. Exit 0 only if every verdict is the expected one.
#
# Block A runs against a local fixture: the request must go out with no Authorization and no Cookie,
# and a region document must round-trip. Block B runs against the live public endpoint of the shared
# canvas: a fresh snapshot verifies identical, a record with one edited cell is caught, and the image
# endpoint must return a PNG of the size asked for.
set -eu
DIR=$(cd "$(dirname "$0")" && pwd)
TOOL="$DIR/pixel-census.py"
W=$(mktemp -d)
trap 'pkill -f "[h]ttp.server 8911" 2>/dev/null || true; rm -rf "$W"' EXIT
fail=0
say() { echo "      $1"; }

expect() { # expect <wanted-exit> <label> <cmd...>
  want=$1; label=$2; shift 2
  set +e; "$@" > "$W/out" 2>&1; got=$?; set -e
  if [ "$got" = "$want" ]; then echo "ok    exit $got  $label"; else echo "FAIL  exit $got (want $want)  $label"; fail=1; fi
  echo "      $(tail -2 "$W/out" | tr '\n' ' ' | cut -c1-150)"
}

# ------------------------------------------------------------------ A. local fixture: an anonymous read
mkdir -p "$W/fixture"
python3 - "$W/fixture" <<'PY'
import json, sys
d = {"world_id": "main", "as_of_seq": "1000", "x": 0, "y": 0, "w": 2, "h": 2,
     "cells": [{"x": 0, "y": 0, "color": 9, "agent_id": "aaaa", "clan_id": None},
               {"x": 1, "y": 0, "color": 0, "agent_id": None, "clan_id": None},
               {"x": 0, "y": 1, "color": 3, "agent_id": "bbbb", "clan_id": "clan-1"},
               {"x": 1, "y": 1, "color": 0, "agent_id": None, "clan_id": None}]}
json.dump(d, open(sys.argv[1] + "/region", "w"))
PY
cat > "$W/fixture/server.py" <<'PY'
import http.server, sys
class H(http.server.BaseHTTPRequestHandler):
    def do_GET(self):
        with open(sys.argv[2], "a") as fh:
            fh.write(self.path + "\n" + str(sorted(self.headers.items())) + "\n")
        body = open(sys.argv[1] + "/region", "rb").read()
        self.send_response(200); self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body))); self.end_headers(); self.wfile.write(body)
    def log_message(self, *a): pass
http.server.HTTPServer(("127.0.0.1", 8911), H).serve_forever()
PY
python3 "$W/fixture/server.py" "$W/fixture" "$W/headers" >/dev/null 2>&1 &
sleep 2

echo "A. a local fixture: the read goes out anonymously and round-trips"
PIXEL_CENSUS_API=http://127.0.0.1:8911/api/agent-world/v1/worlds/main python3 "$TOOL" snapshot 0 0 2 2 -o "$W/f.json" > "$W/out1" 2>&1 || fail=1
if grep -qi 'authorization\|cookie' "$W/headers"; then echo "FAIL  a credential header left the process"; fail=1; else echo "ok          no Authorization and no Cookie in the request headers"; fi
say "$(grep -E 'cells |painters|clans' "$W/out1" | tr -s ' ' | tr '\n' ' ')"
[ "$(grep -c . "$W/f.json" 2>/dev/null || echo 0)" -gt 0 ] || fail=1
expect 0 "the same fixture re-read: identical" \
  env PIXEL_CENSUS_API=http://127.0.0.1:8911/api/agent-world/v1/worlds/main python3 "$TOOL" verify "$W/f.json"
python3 - "$W/f.json" "$W/f-tampered.json" <<'PY'
import json, sys
d = json.load(open(sys.argv[1]))
for c in d["cells"]:
    if c["color"]:
        c["agent_id"] = "someone-else"
        break
json.dump(d, open(sys.argv[2], "w"))
PY
expect 1 "a record whose cell names another author: CHANGED, and the cell is named" \
  env PIXEL_CENSUS_API=http://127.0.0.1:8911/api/agent-world/v1/worlds/main python3 "$TOOL" verify "$W/f-tampered.json"
printf '{"world_id":"main","x":0,"y":0,"w":1,"h":1}' > "$W/not-region.json"
expect 2 "a file that is not a region document: refused" python3 "$TOOL" verify "$W/not-region.json"

# ------------------------------------------------------------------ B. the live public endpoint
echo "B. the live canvas: the public endpoint, anonymously"
expect 0 "a fresh snapshot of the harbour rectangle verifies identical" \
  sh -c "python3 '$TOOL' snapshot 512 512 64 64 -o '$W/live.json' >/dev/null && python3 '$TOOL' verify '$W/live.json'"
expect 0 "the cell census is printed for anyone to compare" \
  sh -c "python3 '$TOOL' snapshot 512 512 64 64 -o '$W/live2.json' | grep -q 'distinct agent_id'"
expect 0 "the image endpoint returns a PNG of the size asked for" \
  python3 "$TOOL" image 520 518 64 64 -o "$W/harbour.png"
expect 2 "a region outside the world is reported, not guessed" \
  python3 "$TOOL" snapshot 99999 99999 4 4 -o "$W/oob.json"

[ "$fail" = 0 ] && { echo "ALL VERDICTS AS EXPECTED"; exit 0; }
echo "SOME VERDICTS DIFFER"; exit 1
